"""Task event publishing service for TaskMaster API.

This module provides event publishing capabilities for task lifecycle events.
Events are published for task creation, updates, and deletion to support
notifications, activity logging, analytics, and webhook triggers.
"""

import logging
from datetime import datetime
from typing import Any, Dict
from uuid import UUID


logger = logging.getLogger(__name__)


class TaskEventPublisher:
    """Publisher for task lifecycle events.

    This service publishes structured events when tasks are created, updated,
    or deleted. Events are published through structured logging as a placeholder
    until a real message queue is integrated.

    All events include event_type, task_id, user_id, timestamp, and changes fields.
    """

    def __init__(self) -> None:
        """Initialize the TaskEventPublisher."""
        self.logger = logger

    def publish_task_created(
        self,
        task_id: UUID,
        user_id: UUID,
        changes: Dict[str, Any],
    ) -> None:
        """Publish a task.created event.

        This event should be published after a task is successfully created
        and committed to the database.

        Args:
            task_id: UUID of the created task.
            user_id: UUID of the user who created the task.
            changes: Dictionary containing task creation details.

        Example:
            >>> publisher.publish_task_created(
            ...     task_id=task.id,
            ...     user_id=task.owner_id,
            ...     changes={"title": "New Task", "status": "pending"}
            ... )
        """
        self._publish_event(
            event_type="task.created",
            task_id=task_id,
            user_id=user_id,
            changes=changes,
        )

    def publish_task_updated(
        self,
        task_id: UUID,
        user_id: UUID,
        changes: Dict[str, Any],
    ) -> None:
        """Publish a task.updated event.

        This event should be published after a task is successfully updated
        and committed to the database.

        Args:
            task_id: UUID of the updated task.
            user_id: UUID of the user who updated the task.
            changes: Dictionary containing the changed fields with old and new values.

        Example:
            >>> publisher.publish_task_updated(
            ...     task_id=task.id,
            ...     user_id=current_user_id,
            ...     changes={
            ...         "title": {"old": "Old Title", "new": "New Title"},
            ...         "status": {"old": "pending", "new": "completed"}
            ...     }
            ... )
        """
        self._publish_event(
            event_type="task.updated",
            task_id=task_id,
            user_id=user_id,
            changes=changes,
        )

    def publish_task_deleted(
        self,
        task_id: UUID,
        user_id: UUID,
        changes: Dict[str, Any],
    ) -> None:
        """Publish a task.deleted event.

        This event should be published after a task is successfully deleted
        and committed to the database.

        Args:
            task_id: UUID of the deleted task.
            user_id: UUID of the user who deleted the task.
            changes: Dictionary containing task deletion details.

        Example:
            >>> publisher.publish_task_deleted(
            ...     task_id=task.id,
            ...     user_id=current_user_id,
            ...     changes={"deleted": True}
            ... )
        """
        self._publish_event(
            event_type="task.deleted",
            task_id=task_id,
            user_id=user_id,
            changes=changes,
        )

    def _publish_event(
        self,
        event_type: str,
        task_id: UUID,
        user_id: UUID,
        changes: Dict[str, Any],
    ) -> None:
        """Internal helper method to publish events.

        This method constructs the event payload with required fields and
        publishes it through structured logging. When a real message queue
        is integrated, this method will be updated to publish to the queue.

        Args:
            event_type: Type of event (task.created, task.updated, task.deleted).
            task_id: UUID of the task.
            user_id: UUID of the user who triggered the event.
            changes: Dictionary containing event-specific change details.
        """
        try:
            payload = {
                "event_type": event_type,
                "task_id": str(task_id),
                "user_id": str(user_id),
                "timestamp": datetime.utcnow().isoformat(),
                "changes": changes,
            }

            # Placeholder: Log structured event until message queue is integrated
            self.logger.info(
                f"Task event published: {event_type}",
                extra={
                    "event_payload": payload,
                    "event_type": event_type,
                    "task_id": str(task_id),
                    "user_id": str(user_id),
                },
            )

        except Exception as exc:
            # Event publishing failures should not break the application
            self.logger.exception(
                f"Failed to publish {event_type} event",
                exc_info=exc,
                extra={
                    "task_id": str(task_id),
                    "user_id": str(user_id),
                },
            )
